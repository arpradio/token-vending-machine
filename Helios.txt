spending token_vending

struct ShareHolder {
    address: PubKeyHash
    percentage: Int
}

struct TokenPrice {
    asset_class: AssetClass
    price_lovelace: Int
}

struct Datum {
    token_prices: []TokenPrice
    shareholders: []ShareHolder
    nonce: Int
    
    func get_price(self, asset_class: AssetClass) -> Int {
        self.token_prices.find((price: TokenPrice) -> Bool {
            price.asset_class == asset_class
        }).price_lovelace
    }
    
    func validate_payment_distribution(self, tx: Tx) -> Bool {
        self.shareholders.all((holder: ShareHolder) -> Bool {
            percentage_amount: Int = holder.percentage;
            expected_lovelace: Int = tx.outputs.fold((acc: Int, output: TxOutput) -> Int {
                if (output.address.credential.switch {
                    pk: PubKey => pk.hash == holder.address,
                    _ => false
                }) {
                    acc
                } else {
                    acc + output.value.get_lovelace() * percentage_amount / 100
                }
            }, 0);
            
            tx.value_sent_to(holder.address).get_lovelace() >= expected_lovelace
        })
    }
}

enum Redeemer {
    Buy {
        buyer: PubKeyHash
    }
    Withdraw {
        owner: PubKeyHash
    }
}

func main(datum: Datum, redeemer: Redeemer, ctx: ScriptContext) -> Bool {
    tx: Tx = ctx.tx;
    
    redeemer.switch {
        b: Buy => {
            script_address: ValidatorHash = ctx.get_current_validator_hash();
            
            purchased_assets: Value = tx.value_locked_by(script_address).get_assets();
            
            buyer_payment: Value = tx.value_sent_to(b.buyer);
            
            all_prices_correct: Bool = purchased_assets.to_map().fold((acc: Bool, mph: MintingPolicyHash, tokens: Map[ByteArray]Int) -> Bool {
                tokens.fold((inner_acc: Bool, token_name: ByteArray, amount: Int) -> Bool {
                    asset: AssetClass = AssetClass::new(mph, token_name);
                    price: Int = datum.get_price(asset);
                    
                    inner_acc && price * amount <= buyer_payment.get_lovelace()
                }, acc)
            }, true);
            
            datum.validate_payment_distribution(tx) && 
            all_prices_correct &&
            tx.is_signed_by(b.buyer)
        },
        w: Withdraw => {
            tx.is_signed_by(w.owner) &&
            datum.shareholders.any((holder: ShareHolder) -> Bool {
                holder.address == w.owner
            })
        }
    }
}
